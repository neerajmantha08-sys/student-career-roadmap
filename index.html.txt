    <script>
        // State Management
        let currentRoadmap = null;
        let pendingDeleteId = null;

        // Navigation Controller
        function navigateTo(viewName) {
            const views = ['home', 'builder', 'processing', 'roadmap', 'saved'];
            views.forEach(v => {
                const el = document.getElementById(`view-${v}`);
                if (el) el.classList.add('hidden');
            });

            const activeView = document.getElementById(`view-${viewName}`);
            if (activeView) activeView.classList.remove('hidden');

            // Nav Tabs styling
            ['home', 'builder', 'saved'].forEach(tab => {
                const navBtn = document.getElementById(`nav-${tab}`);
                if (navBtn) {
                    if (tab === viewName) {
                        navBtn.className = "px-3 py-2 rounded-lg text-sm font-semibold text-brand-700 bg-brand-50 transition-colors";
                    } else {
                        navBtn.className = "px-3 py-2 rounded-lg text-sm font-semibold text-slate-600 hover:text-slate-900 hover:bg-slate-100 transition-colors";
                    }
                }
            });

            if (viewName === 'saved') {
                renderSavedRoadmaps();
            }
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Local Storage Utilities
        function getSavedRoadmaps() {
            try {
                const data = localStorage.getItem('career_roadmaps');
                return data ? JSON.parse(data) : [];
            } catch (e) {
                console.error("Failed to read roadmaps from localStorage", e);
                return [];
            }
        }

        function saveRoadmapsList(list) {
            try {
                localStorage.setItem('career_roadmaps', JSON.stringify(list));
                updateSavedBadge();
            } catch (e) {
                console.error("Failed to save roadmaps to localStorage", e);
            }
        }

        function updateSavedBadge() {
            const list = getSavedRoadmaps();
            const badge = document.getElementById('saved-count-badge');
            if (badge) badge.textContent = list.length;
        }

        // Save Current Active Roadmap
        function saveCurrentRoadmap() {
            if (!currentRoadmap) return;

            let savedList = getSavedRoadmaps();
            const existingIndex = savedList.findIndex(item => item.id === currentRoadmap.id);

            if (existingIndex >= 0) {
                savedList[existingIndex] = currentRoadmap;
            } else {
                savedList.unshift(currentRoadmap);
            }

            saveRoadmapsList(savedList);

            // Update UI button text
            const saveBtnText = document.getElementById('save-btn-text');
            if (saveBtnText) saveBtnText.textContent = "Saved!";
            setTimeout(() => {
                if (saveBtnText) saveBtnText.textContent = "Save Roadmap";
            }, 2000);
        }

        // Handle Form Submission & AI Agent Simulation
        function handleFormSubmit(e) {
            e.preventDefault();

            const studentName = document.getElementById('input-name').value.trim();
            const degree = document.getElementById('input-year').value.trim();
            const targetRole = document.getElementById('input-role').value.trim();
            const domain = document.getElementById('input-domain').value.trim();
            const skills = document.getElementById('input-skills').value.trim();
            const hours = document.getElementById('input-hours').value;
            const timeframe = document.getElementById('input-timeframe').value;

            navigateTo('processing');

            const steps = [
                { title: "Analyzing Student Profile...", desc: "Extracting core strengths and technical readiness.", progress: "25%" },
                { title: "Performing Skill Gap Analysis...", desc: "Evaluating skills against industry benchmarks.", progress: "50%" },
                { title: "Designing Portfolio Projects...", desc: "Crafting real-world project recommendations.", progress: "75%" },
                { title: "Finalizing Step-by-Step Milestones...", desc: "Building weekly timeline plan.", progress: "100%" }
            ];

            let stepIndex = 0;
            const stepInterval = setInterval(() => {
                stepIndex++;
                if (stepIndex < steps.length) {
                    document.getElementById('processing-step-title').textContent = steps[stepIndex].title;
                    document.getElementById('processing-step-desc').textContent = steps[stepIndex].desc;
                    document.getElementById('processing-bar').style.width = steps[stepIndex].progress;
                } else {
                    clearInterval(stepInterval);
                    // Generate AI Data Object
                    currentRoadmap = generateAIModelOutput(studentName, degree, targetRole, domain, skills, hours, timeframe);
                    
                    // Auto save to local storage immediately
                    saveCurrentRoadmap();

                    // Render and display
                    displayRoadmap(currentRoadmap);
                    navigateTo('roadmap');
                }
            }, 700);
        }

        function renderSavedRoadmaps() {
            const list = getSavedRoadmaps();
            const grid = document.getElementById('saved-cards-grid');
            const emptyState = document.getElementById('saved-empty-state');

            updateSavedBadge();

            if (list.length === 0) {
                grid.innerHTML = '';
                emptyState.classList.remove('hidden');
                return;
            }

            emptyState.classList.add('hidden');
            grid.innerHTML = list.map(item => `
                <div class="glass-card p-6 rounded-2xl border border-slate-200 shadow-sm hover:shadow-md transition flex flex-col justify-between space-y-4">
                    <div class="space-y-3">
                        <div class="flex justify-between items-start">
                            <div>
                                <h3 class="font-extrabold text-slate-900 text-lg">${item.targetRole}</h3>
                                <p class="text-xs font-semibold text-slate-500">${item.studentName}</p>
                            </div>
                            <span class="px-2.5 py-1 rounded-full bg-brand-100 text-brand-800 text-xs font-bold">${item.timeframe}</span>
                        </div>
                        <p class="text-xs text-slate-600 line-clamp-2">${item.summary}</p>
                        <div class="text-[11px] text-slate-400 font-medium">Created: ${item.dateStr}</div>
                    </div>
                    <div class="flex items-center space-x-2 pt-2 border-t border-slate-100">
                        <button onclick="viewSavedRoadmap('${item.id}')" class="flex-1 bg-brand-600 hover:bg-brand-700 text-white font-bold py-2 rounded-xl text-xs transition text-center">
                            View Roadmap
                        </button>
                        <button onclick="openDeleteModal('${item.id}')" class="p-2 text-slate-400 hover:text-red-600 hover:bg-red-50 rounded-xl transition">
                            <i class="fa-regular fa-trash-can text-sm"></i>
                        </button>
                    </div>
                </div>
            `).join('');
        }

        function viewSavedRoadmap(id) {
            const list = getSavedRoadmaps();
            const found = list.find(item => item.id === id);
            if (found) {
                currentRoadmap = found;
                displayRoadmap(currentRoadmap);
                navigateTo('roadmap');
            }
        }

        // Delete Modal Handler
        function openDeleteModal(id) {
            pendingDeleteId = id;
            document.getElementById('delete-modal').classList.remove('hidden');
        }

        function closeDeleteModal() {
            pendingDeleteId = null;
            document.getElementById('delete-modal').classList.add('hidden');
        }

        document.getElementById('confirm-delete-btn').addEventListener('click', () => {
            if (pendingDeleteId) {
                let list = getSavedRoadmaps();
                list = list.filter(item => item.id !== pendingDeleteId);
                saveRoadmapsList(list);
                renderSavedRoadmaps();
                closeDeleteModal();
            }
        });

        // Export PDF Functionality
        function exportPDF() {
            const element = document.getElementById('roadmap-content');
            if (!element) return;

            if (typeof html2pdf !== 'undefined') {
                const opt = {
                    margin:       0.3,
                    filename:     `${currentRoadmap ? currentRoadmap.studentName.replace(/\s+/g, '_') : 'Career'}_Roadmap.pdf`,
                    image:        { type: 'jpeg', quality: 0.98 },
                    html2canvas:  { scale: 2 },
                    jsPDF:        { unit: 'in', format: 'letter', orientation: 'portrait' }
                };
                html2pdf().set(opt).from(element).save();
            } else {
                window.print();
            }
        }
